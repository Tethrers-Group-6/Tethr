document.getElementById('register-form').addEventListener('submit', async (e) => { e.preventDefault(); const box = document.getElementById('form-error'); box.hidden = true; const res = await fetch('/register', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ display_name: document.getElementById('display-name').value, email: document.getElementById('email').value, password: document.getElementById('password').value }) }); const data = await res.json().catch(() => ({})); if (!res.ok) { box.textContent = data.error || 'error here.'; box.hidden = false; return; } window.location.href = '/login'; });
